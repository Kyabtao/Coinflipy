📖 Fairness Formula

makerHash  = SHA256(makerSeed : gameId : amount : side)
takerHash  = SHA256(takerSeed : gameId : amount : side)
commit     = SHA256(serverSeed)        # published pre-flip
combined   = BigInt(makerHash) + BigInt(takerHash) + BigInt(commit)
finalHash  = SHA256(combined.toString(16) + ':' + gameId)
firstByte  = parseInt(finalHash.slice(0,2), 16)
result     = firstByte % 2 == 0 ? HEADS : TAILS
jackpot    = firstByte == 0  (1 in 256, only when pool armed)
